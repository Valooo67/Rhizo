cmake_minimum_required(VERSION 3.22)
project(RHIZO VERSION 0.2.0 LANGUAGES C CXX)
if(NOT ANDROID)
 message(FATAL_ERROR "RHIZO runtime is Android ARM64 only. Configure with the Android NDK.")
endif()
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
 message(FATAL_ERROR "RHIZO supports arm64-v8a only; received: ${ANDROID_ABI}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(GLSLC glslc REQUIRED)
set(RHIZO_SHADER_DIR ${CMAKE_CURRENT_BINARY_DIR}/shaders)
set(RHIZO_SHADER_HEADER ${RHIZO_SHADER_DIR}/shader_spv.hpp)
add_custom_command(OUTPUT ${RHIZO_SHADER_HEADER}
 COMMAND ${CMAKE_COMMAND} -E make_directory ${RHIZO_SHADER_DIR}
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/pbr.vert -o ${RHIZO_SHADER_DIR}/pbr.vert.spv
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/pbr.frag -o ${RHIZO_SHADER_DIR}/pbr.frag.spv
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/pbr_instanced.vert -o ${RHIZO_SHADER_DIR}/pbr_instanced.vert.spv
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/composite.vert -o ${RHIZO_SHADER_DIR}/composite.vert.spv
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/composite.frag -o ${RHIZO_SHADER_DIR}/composite.frag.spv
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/shadow.vert -o ${RHIZO_SHADER_DIR}/shadow.vert.spv
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/skybox.vert -o ${RHIZO_SHADER_DIR}/skybox.vert.spv
 COMMAND ${GLSLC} -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/skybox.frag -o ${RHIZO_SHADER_DIR}/skybox.frag.spv
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_spirv.py ${RHIZO_SHADER_HEADER} ${RHIZO_SHADER_DIR}/pbr.vert.spv ${RHIZO_SHADER_DIR}/pbr.frag.spv ${RHIZO_SHADER_DIR}/pbr_instanced.vert.spv ${RHIZO_SHADER_DIR}/composite.vert.spv ${RHIZO_SHADER_DIR}/composite.frag.spv ${RHIZO_SHADER_DIR}/shadow.vert.spv ${RHIZO_SHADER_DIR}/skybox.vert.spv ${RHIZO_SHADER_DIR}/skybox.frag.spv
 DEPENDS shaders/pbr.vert shaders/pbr.frag shaders/pbr_instanced.vert shaders/composite.vert shaders/composite.frag shaders/shadow.vert shaders/skybox.vert shaders/skybox.frag tools/embed_spirv.py VERBATIM)
add_custom_target(rhizo_shaders DEPENDS ${RHIZO_SHADER_HEADER})
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_FBX_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_OBJ_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_COLLADA_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(assimp GIT_REPOSITORY https://github.com/assimp/assimp.git GIT_TAG v5.4.3 GIT_SHALLOW TRUE)
set(BUILD_BULLET2_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_BULLET3 OFF CACHE BOOL "" FORCE)
set(BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_PYBULLET OFF CACHE BOOL "" FORCE)
set(BUILD_OPENGL3_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_CPU_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_BULLET3_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_DOUBLE_PRECISION OFF CACHE BOOL "" FORCE)
set(INSTALL_LIBS OFF CACHE BOOL "" FORCE)
set(BULLET2_MULTITHREADING OFF CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
FetchContent_Declare(bullet3 GIT_REPOSITORY https://github.com/bulletphysics/bullet3.git GIT_TAG 3.25 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(bullet3)
FetchContent_MakeAvailable(assimp)
find_library(ANDROID_VULKAN_LIBRARY vulkan REQUIRED)
find_library(ANDROID_LOG_LIBRARY log)
find_library(ANDROID_LIBRARY android REQUIRED)
find_library(ANDROID_Z_LIBRARY z REQUIRED)
add_library(rhizo SHARED src/core/rhizo.cpp src/vulkan/vulkan_context.cpp src/vulkan/viewport.cpp src/vulkan/gpu_buffer.cpp src/vulkan/frame_texture.cpp src/vulkan/image_decode.cpp src/vulkan/shadow_map.cpp src/scene/scene.cpp src/scene/scene_io.cpp src/scene/resource_manager.cpp src/scene/animation.cpp src/scene/model_importer.cpp src/scene/editor_geometry.cpp src/physics/physics.cpp)
add_dependencies(rhizo rhizo_shaders)
target_include_directories(rhizo PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${RHIZO_SHADER_DIR})
target_link_libraries(rhizo PRIVATE ${ANDROID_VULKAN_LIBRARY} ${ANDROID_Z_LIBRARY} ${ANDROID_LIBRARY} assimp BulletDynamics BulletCollision LinearMath)
if(ANDROID_LOG_LIBRARY)
 target_link_libraries(rhizo PRIVATE ${ANDROID_LOG_LIBRARY})
endif()
target_compile_options(rhizo PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wno-unused-parameter -Wno-missing-field-initializers)
target_compile_definitions(rhizo PRIVATE VK_USE_PLATFORM_ANDROID_KHR=1 RHIZO_BUILDING_LIBRARY=1)
